cmake_minimum_required(VERSION 3.16)
project(JericoBookshelfCppLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
add_library(jerico_bookshelf_core src/core.cpp src/industrial.cpp src/fracture.cpp src/furniture.cpp src/evaluate.cpp src/materials.cpp src/damage.cpp src/wood.cpp src/damp_wood.cpp src/localized_damage.cpp src/fine_wear.cpp src/interfaces.cpp src/png.cpp src/export_glb.cpp src/export_obj.cpp src/dock_geometry.cpp src/dock_deform.cpp src/dock_materials.cpp src/bookshelf.cpp)
target_include_directories(jerico_bookshelf_core PUBLIC include third_party)
target_link_libraries(jerico_bookshelf_core PUBLIC ZLIB::ZLIB)
if(MSVC)
 target_compile_options(jerico_bookshelf_core PRIVATE /W4 /fp:strict)
else()
 target_compile_options(jerico_bookshelf_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
enable_testing()
foreach(t bookshelf_tests dock_deform_tests dock_geometry_tests)
 add_executable(${t} tests/${t}.cpp)
 target_link_libraries(${t} PRIVATE jerico_bookshelf_core)
 add_test(NAME ${t} COMMAND ${t})
endforeach()
target_sources(jerico_bookshelf_core PRIVATE src/bookshelf_materials.cpp)
add_executable(bookshelf_material_tests tests/bookshelf_material_tests.cpp)
target_link_libraries(bookshelf_material_tests PRIVATE jerico_bookshelf_core)
add_test(NAME bookshelf_material_tests COMMAND bookshelf_material_tests ${CMAKE_BINARY_DIR}/material-tests)
add_executable(jerico_bookshelf src/bookshelf_main.cpp)
target_link_libraries(jerico_bookshelf PRIVATE jerico_bookshelf_core)
add_executable(bookshelf_layout_tests tests/bookshelf_layout_tests.cpp)
target_link_libraries(bookshelf_layout_tests PRIVATE jerico_bookshelf_core)
add_test(NAME bookshelf_layout_tests COMMAND bookshelf_layout_tests ${CMAKE_BINARY_DIR}/layout-tests)
set_tests_properties(bookshelf_layout_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
