cmake_minimum_required(VERSION 3.16)
project(JericoShelfPropsCppLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
add_library(jerico_shelf_props src/core.cpp src/industrial.cpp src/fracture.cpp src/furniture.cpp src/evaluate.cpp src/materials.cpp src/damage.cpp src/wood.cpp src/damp_wood.cpp src/localized_damage.cpp src/fine_wear.cpp src/interfaces.cpp src/png.cpp src/export_glb.cpp src/export_obj.cpp src/shelf_props.cpp src/shelf_props_materials.cpp src/shelf_props_placement.cpp)
target_include_directories(jerico_shelf_props PUBLIC include third_party)
target_link_libraries(jerico_shelf_props PUBLIC ZLIB::ZLIB)
if(MSVC)
 target_compile_options(jerico_shelf_props PRIVATE /W4 /fp:strict)
else()
 target_compile_options(jerico_shelf_props PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
enable_testing()
add_executable(shelf_props_tests tests/shelf_props_tests.cpp)
target_link_libraries(shelf_props_tests PRIVATE jerico_shelf_props)
add_test(NAME shelf_props_tests COMMAND shelf_props_tests ${CMAKE_BINARY_DIR}/test-out)
add_executable(jerico_shelf_props_cli src/shelf_props_main.cpp)
set_target_properties(jerico_shelf_props_cli PROPERTIES OUTPUT_NAME jerico_shelf_props)
target_link_libraries(jerico_shelf_props_cli PRIVATE jerico_shelf_props)
add_executable(placement_tests tests/placement_tests.cpp)
target_link_libraries(placement_tests PRIVATE jerico_shelf_props)
add_test(NAME placement_tests COMMAND placement_tests)
foreach(t png_tests evaluate_tests material_scope_tests)
 add_executable(${t} tests/${t}.cpp)
 target_link_libraries(${t} PRIVATE jerico_shelf_props)
 add_test(NAME ${t} COMMAND ${t})
endforeach()
foreach(t export_tests export_obj_tests)
 add_executable(${t} tests/${t}.cpp)
 target_link_libraries(${t} PRIVATE jerico_shelf_props)
 add_test(NAME ${t} COMMAND ${t} ${CMAKE_BINARY_DIR}/${t}-out)
endforeach()
