cmake_minimum_required(VERSION 3.16)
project(JericoProceduralCppLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(jerico_procedural_core src/core.cpp src/industrial.cpp src/fracture.cpp src/furniture.cpp src/evaluate.cpp src/materials.cpp src/damage.cpp src/wood.cpp src/damp_wood.cpp src/localized_damage.cpp src/fine_wear.cpp src/interfaces.cpp src/png.cpp src/export_glb.cpp src/export_obj.cpp)

find_package(ZLIB REQUIRED)
target_link_libraries(jerico_procedural_core PUBLIC ZLIB::ZLIB)
if(MSVC)
 target_compile_options(jerico_procedural_core PRIVATE /W4 /fp:strict)
else()
 target_compile_options(jerico_procedural_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
add_executable(jerico_procedural src/main.cpp)
target_link_libraries(jerico_procedural PRIVATE jerico_procedural_core)
enable_testing()
