cmake_minimum_required(VERSION 3.16)
project(JericoCobwebUVClipProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cobweb_uv_clip src/uv_clip.cpp)
target_include_directories(cobweb_uv_clip PUBLIC include)
if(MSVC)
  target_compile_options(cobweb_uv_clip PRIVATE /W4)
else()
  target_compile_options(cobweb_uv_clip PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(uv_clip_tests tests/uv_clip_tests.cpp)
target_link_libraries(uv_clip_tests PRIVATE cobweb_uv_clip)
target_compile_definitions(uv_clip_tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
enable_testing()
add_test(NAME uv_clip_contract COMMAND uv_clip_tests)
