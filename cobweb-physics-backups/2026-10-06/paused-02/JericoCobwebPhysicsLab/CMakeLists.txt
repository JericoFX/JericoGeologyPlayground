cmake_minimum_required(VERSION 3.16)
project(JericoCobwebLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cobweb_core src/geometry.cpp src/fibers.cpp src/bake.cpp src/damage.cpp src/export.cpp)
target_include_directories(cobweb_core PUBLIC include)
if(MSVC)
 target_compile_options(cobweb_core PRIVATE /W4)
else()
 target_compile_options(cobweb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(cobweb_lab src/main.cpp)
target_link_libraries(cobweb_lab PRIVATE cobweb_core)

add_executable(cobweb_core_tests tests/core_tests.cpp)
target_link_libraries(cobweb_core_tests PRIVATE cobweb_core)
enable_testing()
add_test(NAME core_geometry COMMAND cobweb_core_tests)

add_library(cobweb_physics src/physics_graph.cpp src/tear_binding.cpp src/xpbd_solver.cpp)
target_link_libraries(cobweb_physics PUBLIC cobweb_core)
target_compile_options(cobweb_physics PRIVATE -Wall -Wextra -Wpedantic)
add_executable(physics_binding_tests tests/physics_binding_tests.cpp)
target_link_libraries(physics_binding_tests PRIVATE cobweb_physics)
add_test(NAME physics_binding COMMAND physics_binding_tests)
add_executable(cobweb_sim src/sim_cli.cpp)
target_link_libraries(cobweb_sim PRIVATE cobweb_physics)
add_executable(physics_solver_tests tests/physics_solver_tests.cpp)
target_link_libraries(physics_solver_tests PRIVATE cobweb_physics)
add_test(NAME physics_solver COMMAND physics_solver_tests)

add_executable(physics_panel_tests tests/physics_panel_tests.cpp)
target_link_libraries(physics_panel_tests PRIVATE cobweb_physics)
add_test(NAME physics_panel_endurance COMMAND physics_panel_tests)

add_library(cobweb_wind src/wind.cpp)
target_link_libraries(cobweb_wind PUBLIC cobweb_physics)
add_executable(physics_wind_tests tests/physics_wind_tests.cpp)
target_link_libraries(physics_wind_tests PRIVATE cobweb_wind)
add_test(NAME physics_wind COMMAND physics_wind_tests)
