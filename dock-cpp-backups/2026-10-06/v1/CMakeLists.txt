cmake_minimum_required(VERSION 3.16)
project(JericoDockCppLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
add_library(jerico_dock_core src/core.cpp src/industrial.cpp src/fracture.cpp src/furniture.cpp src/evaluate.cpp src/materials.cpp src/damage.cpp src/wood.cpp src/damp_wood.cpp src/localized_damage.cpp src/fine_wear.cpp src/interfaces.cpp src/png.cpp src/export_glb.cpp src/export_obj.cpp src/dock.cpp)
target_include_directories(jerico_dock_core PUBLIC include third_party)
target_link_libraries(jerico_dock_core PUBLIC ZLIB::ZLIB)
if(MSVC)
 target_compile_options(jerico_dock_core PRIVATE /W4 /fp:strict)
else()
 target_compile_options(jerico_dock_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
enable_testing()
add_executable(dock_tests tests/dock_tests.cpp)
target_link_libraries(dock_tests PRIVATE jerico_dock_core)
add_test(NAME dock_tests COMMAND dock_tests)
target_sources(jerico_dock_core PRIVATE src/dock_geometry.cpp)
target_sources(jerico_dock_core PRIVATE src/dock_materials.cpp)
target_sources(jerico_dock_core PRIVATE src/dock_deform.cpp)
add_executable(dock_deform_tests tests/dock_deform_tests.cpp)
target_link_libraries(dock_deform_tests PRIVATE jerico_dock_core)
add_test(NAME dock_deform_tests COMMAND dock_deform_tests)
add_executable(dock_material_tests tests/dock_material_tests.cpp)
target_link_libraries(dock_material_tests PRIVATE jerico_dock_core)
add_test(NAME dock_material_tests COMMAND dock_material_tests ${CMAKE_BINARY_DIR}/material-tests)
add_executable(dock_geometry_tests tests/dock_geometry_tests.cpp)
target_link_libraries(dock_geometry_tests PRIVATE jerico_dock_core)
add_test(NAME dock_geometry_tests COMMAND dock_geometry_tests)
add_executable(jerico_dock src/dock_main.cpp)
target_link_libraries(jerico_dock PRIVATE jerico_dock_core)
