cmake_minimum_required(VERSION 3.16)
project(JericoCobwebLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cobweb_core src/geometry.cpp src/fibers.cpp src/bake.cpp src/export.cpp)
target_include_directories(cobweb_core PUBLIC include)
if(MSVC)
 target_compile_options(cobweb_core PRIVATE /W4)
else()
 target_compile_options(cobweb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(cobweb_lab src/main.cpp)
target_link_libraries(cobweb_lab PRIVATE cobweb_core)

add_executable(cobweb_core_tests tests/core_tests.cpp)
target_link_libraries(cobweb_core_tests PRIVATE cobweb_core)
enable_testing()
add_test(NAME core_geometry COMMAND cobweb_core_tests)
