cmake_minimum_required(VERSION 3.16)
project(CobwebImpactLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cobweb_asset src/geometry.cpp src/fibers.cpp src/bake.cpp src/export.cpp)
target_include_directories(cobweb_asset PUBLIC include)
add_library(cobweb_impact src/impact.cpp)
target_link_libraries(cobweb_impact PUBLIC cobweb_asset)
add_executable(impact_demo src/impact_cli.cpp)
target_link_libraries(impact_demo PRIVATE cobweb_impact)
add_executable(impact_tests tests/impact_tests.cpp)
target_link_libraries(impact_tests PRIVATE cobweb_impact)
enable_testing()
add_test(NAME impact_core COMMAND impact_tests)
add_executable(impact_benchmark tests/impact_benchmark.cpp)
target_link_libraries(impact_benchmark PRIVATE cobweb_impact)
