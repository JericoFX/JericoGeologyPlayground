cmake_minimum_required(VERSION 3.25)
project(JEGeometryAnalysisStandalone LANGUAGES CXX)
get_filename_component(PLAYGROUND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
include("${PLAYGROUND_ROOT}/CMake/GeometryAnalysis.cmake")
option(JE_GEOMETRY_ASAN "Instrument this standalone consumer with AddressSanitizer" OFF)
if(MSVC)
    if(JE_GEOMETRY_ASAN)
        target_compile_options(JEGeometryAnalysis PRIVATE /fsanitize=address /Zi)
        target_compile_options(GeometryAnalysisTests PRIVATE /fsanitize=address /Zi)
        target_compile_options(GeometrySectionsTests PRIVATE /fsanitize=address /Zi)
        target_link_options(GeometryAnalysisTests PRIVATE /INCREMENTAL:NO /DEBUG)
        target_link_options(GeometrySectionsTests PRIVATE /INCREMENTAL:NO /DEBUG)
    endif()
endif()
